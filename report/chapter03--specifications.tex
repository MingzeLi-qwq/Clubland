\section{Functional specifications}
\label{sect:functional-specifications}
\begin{enumerate}
    \def\labelenumi{\arabic{enumi}.}
    \item
      \textbf{User Registration and Authentication System}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Allows users to register new accounts using email and log in with
          a username and password. Includes password reset. Built with
          Django's built-in authentication module for secure access.
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          Homepage avatars provide access to the registration and login
          pages when not logged in, and to the Dashboard for account
          settings once logged in.
        \end{itemize}
      \item
        Role Access:
    
        \begin{itemize}
        \item
          All users must log in to access society club features. Admin
          features require admin role.
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Ensures user data privacy and platform control by avoiding
          third-party login dependencies.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Customizable Club Hub}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Enables club members to personalize club homepages, share their
          ideas with image uploads, notices, event widgets, and clocks using
          React Grid Layout.
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          Click "Club hubs" on the Navbar then select club.
        \end{itemize}
      \item
        Role Access:
    
        \begin{itemize}
        \item
          Every club members can access, view and edit.
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Helps clubs build their identity and promote diversity on the
          platform.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Member Management}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Club Manager can add/remove members and assign roles
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          Club Manager access club detail page and click "Club Manager Page"
          can find membership management function
        \end{itemize}
      \item
        Role Access:
    
        \begin{itemize}
        \item
          Only club manager
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Enhances structural efficiency and supports autonomous society
          club operations.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{News Publishing}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Users can publish news using a rich text editor (Django
          Summernote).
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          Click "News" in the Navbar, then "Publish News"
        \end{itemize}
      \item
        Role Access:
    
        \begin{itemize}
        \item
          Regular users can only post news in their own name, Club Manager
          can post news in the name of the club.
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Improves internal communication and reduces reliance on external
          platforms.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Forum Post and Reply System}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Users can create and reply to forum posts to facilitate
          discussion.
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          Click on "Forums" in the Navbar to access related functions.
        \end{itemize}
      \item
        Role-Based Access:
    
        \begin{itemize}
        \item
          Logged-in users can post and reply in their own name, Club Manager
          can post forum in the name of the club.
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Promotes communication and knowledge sharing.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Multi-type Notification Management}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Supports various notification types like requests status,
          membership update.
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          Notification menu(little bell) in the top navigation bar.
        \end{itemize}
      \item
        Role-Based Access:
    
        \begin{itemize}
        \item
          All logged-in users can receive relevant notifications.
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Enhances timely and relevant communication.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Event Registration System}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          User can RSVP events, check event status, add event into their
          calendar and cancel if needed.
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          User can access by click "Events" in Navbar
        \end{itemize}
      \item
        Role-Based Access:
    
        \begin{itemize}
        \item
          All regular users can access and RSVP events.
        \item
          Club manager can create and modify events from "Club manager page
          - events"
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Ensures orderly and justified creation of new societies.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{New Club Request and Approval}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Users can request to create new clubs. Admins can approve or
          reject applications.
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          "Create Club" button on the bottom of club list page, "admin panel
          - requests" for review.
        \end{itemize}
      \item
        Role Access:
    
        \begin{itemize}
        \item
          Users can apply. Admins can approve/reject.
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Clarifies structure and responsibilities within clubs.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{System Administrator Panel}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          The System Administrator can review New Club Requests, modify or
          delete content (Memberships, News, Events, Clubs, etc.) in the
          system to ensure compliance with policies.
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          After System Administrator login, click "Admin Panel" in Navbar.
        \end{itemize}
      \item
        Role Access:
    
        \begin{itemize}
        \item
          System Administrator Only
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Supports institutional governance and ensures content compliance.
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Search and Filtering Capability}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Most lists within the system provide search and filtering
          functionalities, including:
    
          \begin{itemize}
          \item
            Exploratory lists accessible to general users, such as Clubs,
            Events, News, and Forums.
          \item
            Management-oriented lists available to Club Managers and
            Administrators, such as Memberships and Requests.
          \end{itemize}
        \end{itemize}
      \item
        How to Access:
    
        \begin{itemize}
        \item
          Search and filtering tools are positioned at the top of most
          tables.
        \item
          For example, in the Event List:
    
          \begin{itemize}
          \item
            After navigating to "Events" via the Navbar, users can find the
            search and filtering tools located above the event listings.
          \end{itemize}
        \end{itemize}
      \item
        Role-Based Access:
    
        \begin{itemize}
        \item
          All users have access to search and filter functionalities,
          although specific searchable content varies based on user
          permissions.
        \end{itemize}
      \item
        Business Goal:
    
        \begin{itemize}
        \item
          Enhance user experience and lower the entry barrier for users to
          easily find relevant information.
        \end{itemize}
      \end{itemize}
    \end{enumerate}

\section{Non-functional specifications}
\label{sect:non-functional-specifications}

\begin{enumerate}
    \def\labelenumi{\arabic{enumi}.}
    \item
      \textbf{Security}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          Ensure the security, confidentiality, integrity and availability
          of user and system data by preventing unauthorization.
        \end{itemize}
      \item
        Specific measures:
    
        \begin{itemize}
        \item
          Passwords are securely stored using Django's built-in
          authentication.
        \item
          CSRF protection is enabled via \texttt{CSRF\_TRUSTED\_ORIGINS}.
        \item
          Using the mixin method in key apps enables code reuse, centralized
          management of permissions, modular security checks, and enhances
          the flexibility and maintainability of the view layer through
          combinatorial inheritance. For example:
    
          \begin{itemize}
          \item
            \texttt{UserTypeRequiredMixin} implements user type whitelisting
            control via the allowed\_types parameter
          \item
            \texttt{LoginRequiredMixin} Harmonizes the handling of
            non-logged-in user redirects.
          \item
            \texttt{ClubManagerRequiredMixin} wraps the club manager
            permission checking logic
          \end{itemize}
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Usability}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          The platform is intuitive, easy to learn and use, with a clear
          interface aligned with user expectations.
        \end{itemize}
      \item
        Specific measures:
    
        \begin{itemize}
        \item
          Cross-application style harmonization through global CSS variables
          and component libraries (e.g. Bootstrap)
        \item
          Three-tier information structure (Navbar → Left menu → Card
          Controls) for Club Manager Page, User Dashboard and Admin Panel to
          reduce the cognitive load on users.
        \item
          Implemented search function in most of the forms to reduce the
          user\textquotesingle s difficulty. For example:
    
          \begin{itemize}
          \item
            Search box at the top of the "Clubs" page
          \end{itemize}
        \end{itemize}
      \end{itemize}
    \item
      \textbf{Extensibility}
    
      \begin{itemize}
      \item
        Feature Description:
    
        \begin{itemize}
        \item
          The system architecture supports seamless addition of new features
          and modules.
        \end{itemize}
      \item
        Specific measures:
    
        \begin{itemize}
        \item
          Each sub-app maintains its own routing configuration, which is
          dynamically integrated through the project\textquotesingle s root
          URLs file, making it easy to extend at a later stage.
        \item
          The left menu bar of the Club Manager Page, User Dashboard and
          Admin Panel is modular and inheritable, allowing for the expansion
          of new control functions.
        \item
          Club Hub uses the Widget design pattern, which can be extended to
          other application scenarios at a later stage.
        \end{itemize}
      \end{itemize}
    \end{enumerate}
